第一步：定义成绩数据（第1次提交） // script.js const scores = [ { name: '李四', score: 92 }, { name: '王五', score: 45 }, { name: '赵六', score: 77 }, { name: '孙七', score: 59 }, { name: '周八', score: 88 }, { name: '吴九', score: 105 }, // 故意混入非法值：满分100 { name: '郑十', score: -3 } // 负分 ]; git add . git commit -m "添加成绩数据定义" git push 检查点1：Console无报错；数据能被console.table(scores)正常打印。 第二步：清洗与统计函数（第2次提交） // 清洗：只保留0至100之间的合法成绩 const cleanScores = (list) => list.filter(s => s.score >= 0 && s.score <= 100); // 平均分 const average = (list) => { if (list.length === 0) return 0; // 空数组保护，除零会产生NaN const total = list.reduce((sum, s) => sum + s.score, 0); return (total / list.length).toFixed(2); }; // 最高分 const highest = (list) => list.reduce((max, s) => s.score > max.score ? s : max, list[0]); // 不及格名单 const failed = (list) => list.filter(s => s.score < 60).map(s => s.name); console.log('清洗后：', cleanScores(scores)); console.log('平均分：', average(cleanScores(scores))); console.log('最高分：', highest(cleanScores(scores))); console.log('不及格：', failed(cleanScores(scores))); 先预测每个输出，再刷新页面到Console验证。 git add . git commit -m "添加清洗与统计函数" git push 检查点2：非法成绩被过滤；平均分保留两位小数；每个函数职责单一。 第三步：等级分布与格式化输出（第3次提交） // 等级判定（单值进单值出，纯函数） const toGrade = (score) => { if (score >= 90) return 'A'; if (score >= 80) return 'B'; if (score >= 70) return 'C'; if (score >= 60) return 'D'; return 'F'; }; // 各等级人数统计 const gradeCount = (list) => { const result = { A: 0, B: 0, C: 0, D: 0, F: 0 }; list.forEach(s => { result[toGrade(s.score)]++; }); return result; }; // 格式化报告 const report = (list) => { const valid = cleanScores(list); if (valid.length === 0) { return '没有有效成绩'; } const dist = gradeCount(valid); return `有效人数${valid.length}人，平均${average(valid)}分，最高${highest(valid).score}分（${highest(valid).name}）； 等级分布：A${dist.A}人 B${dist.B}人 C${dist.C}人 D${dist.D}人 F${dist.F}人； 不及格名单：${failed(valid).join('、') || '无'}`; }; try { console.log(report(scores)); } catch (err) { console.error('报告生成失败：', err.message); } 故意实验：把scores改成空数组[]，观察report输出”没有有效成绩”而不是崩溃——这就是防御性编程。 git add . git commit -m "添加等级分布与格式化报告" git push 检查点3：git log三次提交；输出报告完整格式清晰。案例复现完成。 第八部分 自主实践任务：数据计算小工具 一、任务书 题目：任选场景实现一个输入、计算、输出的小工具。可选场景举例：成绩统计、消费记账、运动数据、图书借阅、绩点计算、个税估算。本阶段用prompt输入、console或页面简单输出均可（不要求界面，下次课补）。 基本要求（必须完成）： 序号 要求 是否达成 1 任选场景，工具解决一个明确的小问题 2 数据用数组加对象组织 3 至少3个职责单一的函数 4 至少使用map、filter、reduce中的两种 5 含非法输入处理（清洗或提示，程序不崩溃） 6 函数命名与注释清晰 7 不少于3次有效Git提交，提交说明具体规范，完成远程推送 8 运行结果完整可复现（输入与输出都有截图） 合计 50 独立研究任务（选做）： 以下任务课堂不讲授，需要你自己查阅官方文档或教程、动手试验后完成。做多少得多少，课上可能抽你讲思路。研究结果写入进度报告。 序号 研究任务 是否达成 1 排序深入研究：查阅Array.prototype.sort文档，弄清比较函数返回值规则与排序稳定性，为工具实现按两个字段先后排序（如先按类别再按数值），用测试数据验证并记录结论 2 正则表达式初探：自学正则表达式基础写法，用正则完成一项数据清洗或校验（如去除多余空格、校验手机号或邮箱格式），在进度报告中逐段解释该正则的含义 3 性能对比实验：分别用for循环和reduce实现同一统计，用console.time测量并对比耗时，在进度报告中记录数据并给出结论
